desafio_01_iluminacao.ino 
// ═══════════════════════════════════════════════════════
// DESAFIO 01 — ILUMINAÇÃO INTELIGENTE
// Sensor PIR + Sensor LDR + LED
// ═══════════════════════════════════════════════════════

// ── Definição dos pinos ──────────────────────────────
const int pinoLED  = 13;   // LED de saída
const int pinoPIR  = 7;    // Sensor de presença PIR
const int pinoLDR  = A0;   // Sensor de luminosidade LDR

// ── Parâmetros configuráveis ─────────────────────────
const int limiarLDR = 500;  // Limiar: abaixo = escuro

// ── Variáveis de estado ──────────────────────────────
int estadoPIR  = 0;
int valorLDR   = 0;

// ────────────────────────────────────────────────────
void setup() {
  pinMode(pinoLED, OUTPUT);
  pinMode(pinoPIR, INPUT);
  // pinoLDR é analógico — não precisa de pinMode
  Serial.begin(9600);
  Serial.println("Sistema iniciado: Iluminação Inteligente");
}

// ────────────────────────────────────────────────────
void loop() {
  // Leitura dos sensores
  estadoPIR = digitalRead(pinoPIR);
  valorLDR  = analogRead(pinoLDR);

  // Condição: movimento E ambiente escuro
  if (estadoPIR == HIGH && valorLDR < limiarLDR) {
    digitalWrite(pinoLED, HIGH);   // Liga o LED
    Serial.println("[ON] Movimento + Escuro → LED LIGADO");
  } else {
    digitalWrite(pinoLED, LOW);    // Desliga o LED
    Serial.println("[OFF] Condição não atendida → LED DESLIGADO");
  }

  delay(200);
}

desafio_02_distancia.ino 
// ═══════════════════════════════════════════════════════
// DESAFIO 02 — DISTÂNCIA E ILUMINAÇÃO
// Sensor Ultrassônico + Potenciômetro + LED
// ═══════════════════════════════════════════════════════

// ── Definição dos pinos ──────────────────────────────
const int pinoTrig = 9;     // HC-SR04 TRIG
const int pinoEcho = 8;     // HC-SR04 ECHO
const int pinoLED  = 13;    // LED de saída
const int pinoPOT  = A0;    // Potenciômetro

// ── Variáveis ────────────────────────────────────────
long  duracao     = 0;
float distancia  = 0;
int   valorPOT   = 0;
float limite     = 0;

// ────────────────────────────────────────────────────
void setup() {
  pinMode(pinoTrig, OUTPUT);
  pinMode(pinoEcho, INPUT);
  pinMode(pinoLED,  OUTPUT);
  Serial.begin(9600);
  Serial.println("Sistema iniciado: Distância e Iluminação");
}

// ── Função: medir distância (cm) ─────────────────────
float medirDistancia() {
  digitalWrite(pinoTrig, LOW);
  delayMicroseconds(2);
  digitalWrite(pinoTrig, HIGH);
  delayMicroseconds(10);
  digitalWrite(pinoTrig, LOW);

  duracao = pulseIn(pinoEcho, HIGH);
  return (duracao * 0.034) / 2.0;
}

// ────────────────────────────────────────────────────
void loop() {
  distancia = medirDistancia();

  // Mapeia potenciômetro → limite em cm (10 a 100 cm)
  valorPOT = analogRead(pinoPOT);
  limite   = map(valorPOT, 0, 1023, 10, 100);

  Serial.print("Distância: ");
  Serial.print(distancia);
  Serial.print(" cm | Limite: ");
  Serial.print(limite);
  Serial.println(" cm");

  // Se distância < limite → LED ligado
  if (distancia < limite) {
    digitalWrite(pinoLED, HIGH);
    Serial.println("[ON] Objeto próximo → LED LIGADO");
  } else {
    digitalWrite(pinoLED, LOW);
    Serial.println("[OFF] Objeto distante → LED DESLIGADO");
  }

  delay(300);
}

desafio_03_ambiente.ino 
// ═══════════════════════════════════════════════════════
// DESAFIO 03 — AMBIENTE INTELIGENTE
// Sensor DHT (Temperatura + Umidade) + LEDs indicadores
// ═══════════════════════════════════════════════════════

// ── Biblioteca necessária: DHT sensor library ────────
#include <DHT.h>

// ── Definição dos pinos ──────────────────────────────
#define PINO_DHT        4    // Pino de dados do DHT
#define TIPO_DHT     DHT11   // ou DHT22

const int pinoLedVerde    = 10;  // Condição adequada
const int pinoLedVermelho = 11;  // Temperatura alta
const int pinoLedAmarelo  = 12;  // Umidade baixa

// ── Limites de referência ────────────────────────────
const float TEMP_MAX = 30.0;   // °C — acima = alerta
const float UMID_MIN = 40.0;   // %  — abaixo = alerta

// ── Instância do sensor ──────────────────────────────
DHT dht(PINO_DHT, TIPO_DHT);

// ── Variáveis de leitura ─────────────────────────────
float temperatura = 0;
float umidade    = 0;

// ────────────────────────────────────────────────────
void setup() {
  pinMode(pinoLedVerde,    OUTPUT);
  pinMode(pinoLedVermelho, OUTPUT);
  pinMode(pinoLedAmarelo,  OUTPUT);
  dht.begin();
  Serial.begin(9600);
  Serial.println("Sistema iniciado: Ambiente Inteligente");
}

// ── Função: desligar todos os LEDs ───────────────────
void desligarTodos() {
  digitalWrite(pinoLedVerde,    LOW);
  digitalWrite(pinoLedVermelho, LOW);
  digitalWrite(pinoLedAmarelo,  LOW);
}

// ────────────────────────────────────────────────────
void loop() {
  temperatura = dht.readTemperature();
  umidade    = dht.readHumidity();

  // Verificação de leitura válida
  if (isnan(temperatura) || isnan(umidade)) {
    Serial.println("Erro na leitura do sensor DHT!");
    desligarTodos();
    delay(2000);
    return;
  }

  Serial.print("Temp: "); Serial.print(temperatura);
  Serial.print(" °C | Umidade: "); Serial.println(umidade);

  desligarTodos();

  bool tempAlta = (temperatura > TEMP_MAX);
  bool umidBaixa = (umidade < UMID_MIN);

  if (!tempAlta && !umidBaixa) {
    digitalWrite(pinoLedVerde, HIGH);     // Condição adequada
    Serial.println("[OK] Ambiente normal");
  } else {
    if (tempAlta)  digitalWrite(pinoLedVermelho, HIGH);
    if (umidBaixa) digitalWrite(pinoLedAmarelo,  HIGH);
    Serial.println("[ALERTA] Condição fora do esperado");
  }

  delay(2000);  // DHT precisa de intervalo mínimo
}

Sistema de Detecção de Presença com Sensor PIR
Neste projeto, foi desenvolvido um sistema de automação voltado para a detecção de movimento em tempo real.

Componentes Utilizados: Placa Arduino Uno, sensor infravermelho passivo (PIR), LED indicador, protoboard, resistor e cabos jumper.

Funcionamento e Código: O programa lê o estado do pino digital conectado ao sensor PIR. Quando o sensor detecta movimentação no ambiente, o sinal é alterado para nível alto (HIGH), acionando o LED como indicador visual.

Aplicações Práticas: Iluminação automática residencial, sistemas de alarme e controle de acesso patrimonial.

▸ OBJETIVO E FUNCIONAMENTO

Sistema de Medição de Proximidade com Sensor Ultrassônico e Potenciômetro
Este experimento demonstra um medidor e alerta de distância com calibração analógica.

Componentes Utilizados: Placa Arduino Uno, sensor ultrassônico de distância (HC-SR04), potenciômetro, LED indicador, protoboard, resistor e cabos jumper.

Funcionamento e Código: O sensor ultrassônico calcula a distância enviando um pulso sonoro pelo pino de disparo (Trig) e cronometrando o tempo de retorno pelo pino de eco (Echo). O potenciômetro conectado à entrada analógica permite calibrar os limiares de acionamento. Ao aproximar um objeto a uma distância menor que o limite definido (distância crítica), o circuito aciona o LED de alerta.

Aplicações Práticas: Sensores de estacionamento veicular, detecção de obstáculos em robótica e monitoramento de níveis em reservatórios.
